// 修復版：精準帶入當前切換的代號進行即時行情更新 window.fetchMarketPrice = async function(type, targetSymbol = null) { const symbolInput = targetSymbol || document.getElementById(`${type}_symbolInput`).value.trim(); if (!symbolInput) return; let querySymbol = symbolInput.toUpperCase(); if (type === 'tw' && !querySymbol.endsWith('.TW')) querySymbol += '.TW'; if (fetchController) fetchController.abort(); fetchController = new AbortController(); document.getElementById('syncStatus').innerText = `正在透過 Vercel 後端查詢 ${querySymbol}...`; try { const apiRes = await fetch(`https://costcal-peach.vercel.app/api/stock?symbol=${querySymbol}`, { signal: fetchController.signal }); if (!apiRes.ok) throw new Error("後端查詢失敗"); const result = await apiRes.json(); if (result.currentPrice) { document.getElementById(`${type}_currentPrice`).value = result.currentPrice; if (result.prevClose) document.getElementById(`${type}_prevClose`).value = result.prevClose; if (type === 'tw') calculateTw(); else calculateUs(); saveData(); document.getElementById('syncStatus').innerText = `已成功代入 ${querySymbol} 行情 (${result.currentPrice})`; } } catch (e) { if (e.name === 'AbortError') return; document.getElementById('syncStatus').innerText = `查詢失敗`; } }; // 修復版：切換組合時確保正確抓取新組合的代號 window.switchStock = async function(type, key) { saveUIToCurrentStock(type); loadStockToUI(type, key); // 直接從對應的資料結構中取出該組合的代號去抓股價，避免 DOM 讀取時間差 const store = type === 'tw' ? cloudData.twStocks : cloudData.usStocks; const targetSymbol = store[key]?.symbol; if (targetSymbol) { await fetchMarketPrice(type, targetSymbol); } };
