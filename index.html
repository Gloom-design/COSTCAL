// 修正後的智慧萬用行情查詢（確保傳給後端的是代號而不是中文） window.fetchMarketPrice = async function(type, targetSymbol = null) { const rawInput = targetSymbol || document.getElementById(`${type}_symbolInput`).value.trim(); if (!rawInput) return; let querySymbol = rawInput.toUpperCase(); if (type === 'tw') { // 如果輸入的是中文名稱，透過字典強制轉成代號 if (twStockDictionary[querySymbol]) { querySymbol = twStockDictionary[querySymbol]; } // 確保台股代號格式正確 (純數字 4 碼再加上 .TW 或 .TWO) if (/^\d{4}$/.test(querySymbol)) { const knownTwo = ["6223", "3105", "3293", "5347", "6515", "8299"]; const suffix = knownTwo.includes(querySymbol) ? '.TWO' : '.TW'; if (!querySymbol.endsWith('.TW') && !querySymbol.endsWith('.TWO')) { querySymbol += suffix; } } else { if (!querySymbol.endsWith('.TW') && !querySymbol.endsWith('.TWO')) { querySymbol += '.TW'; } } } if (fetchController) fetchController.abort(); fetchController = new AbortController(); document.getElementById('syncStatus').innerText = `正在查詢 ${querySymbol}...`; try { let apiRes = await fetch(`https://costcal-peach.vercel.app/api/stock?symbol=${querySymbol}`, { signal: fetchController.signal }); if (!apiRes.ok && querySymbol.endsWith('.TW')) { querySymbol = querySymbol.replace('.TW', '.TWO'); apiRes = await fetch(`https://costcal-peach.vercel.app/api/stock?symbol=${querySymbol}`, { signal: fetchController.signal }); } if (!apiRes.ok) throw new Error("後端查詢失敗"); const result = await apiRes.json(); if (result.currentPrice) { document.getElementById(`${type}_currentPrice`).value = result.currentPrice; if (result.prevClose) document.getElementById(`${type}_prevClose`).value = result.prevClose; if (type === 'tw') calculateTw(); else calculateUs(); saveData(); document.getElementById('syncStatus').innerText = `已成功代入 ${querySymbol} 行情 (${result.currentPrice})`; } } catch (e) { if (e.name === 'AbortError') return; document.getElementById('syncStatus').innerText = `查詢失敗，請檢查代號是否正確`; } }; // 修正切換組別時被強制重置的問題 window.switchStock = async function(type, key) { if (type === 'tw') currentTwKey = key; else currentUsKey = key; loadStockToUI(type, key); const store = type === 'tw' ? cloudData.twStocks : cloudData.usStocks; const targetSymbol = store[key]?.symbol; if (targetSymbol) { await fetchMarketPrice(type, targetSymbol); } };
